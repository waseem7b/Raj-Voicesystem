```python
#!/usr/bin/env python3
"""Pull Vapi call data, normalized for the call-review subsystem.

Filtered to Voicsito-owned stage assistants only (and the monolithic
rollback target). Calls handled by other assistants raise ValueError.

CLI:
    python scripts/fetch_call.py --id <vapi-call-id>
    python scripts/fetch_call.py --last 10
    python scripts/fetch_call.py --since 2026-05-19T00:00:00Z
    python scripts/fetch_call.py --last 25 --filter biz_type=cleaning
    python scripts/fetch_call.py --last 25 --filter ended_reason=customer-ended-call
    python scripts/fetch_call.py --last 25 --filter stage_reached=stage2

Output: JSON list of normalized dicts on stdout.
"""
import argparse
import json
import os
import sys
from datetime import datetime, timezone
from pathlib import Path

import httpx

VAPI_BASE = "https://api.vapi.ai"

VOICSITO_ASSISTANT_IDS = {
    "46e47e95-674a-4624-9029-69bb39c4daf0": "stage1",  # Stage 1 Inbound Texts
    "7b2666e0-154d-4c85-82f0-07b4c51ad264": "stage2",  # Stage 2 Qualify
    "7600c2b6-d669-4f2e-8387-e371b0c193d4": "stage3",  # Stage 3 Book
    "261bf1a0-3527-4fcb-956c-b6e227d0bae1": "monolithic",  # Rollback target
}


def _parse_iso(ts: str) -> datetime:
    """Parse an ISO-8601 timestamp (with optional trailing Z)."""
    if ts.endswith("Z"):
        ts = ts[:-1] + "+00:00"
    return datetime.fromisoformat(ts)


def _normalize(raw: dict) -> dict:
    """Convert a raw Vapi /call response into our normalized shape."""
    assistant_id = raw.get("assistantId")
    if assistant_id not in VOICSITO_ASSISTANT_IDS:
        raise ValueError(
            f"call {raw.get('id')} is not a Voicsito call "
            f"(assistantId={assistant_id})"
        )
    stage = VOICSITO_ASSISTANT_IDS[assistant_id]

    started = raw.get("startedAt")
    ended = raw.get("endedAt")
    duration_s = None
    if started and ended:
        duration_s = int(
            (_parse_iso(ended) - _parse_iso(started)).total_seconds()
        )

    customer = raw.get("customer") or {}
    metadata = raw.get("metadata") or {}

    return {
        "call_id": raw.get("id"),
        "started_at": started,
        "ended_at": ended,
        "duration_s": duration_s,
        "ended_reason": raw.get("endedReason"),
        "assistant_id": assistant_id,
        "assistant_name": (raw.get("assistant") or {}).get("name"),
        "stage_reached": stage,
        "customer_number": customer.get("number"),
        "customer_first_name": customer.get("firstName"),
        "biz_type": metadata.get("biz_type"),
        "summary": raw.get("summary") or "",
        "transcript": raw.get("transcript") or "",
        "recording_url": raw.get("recordingUrl"),
        "raw_metadata": metadata,
    }


def fetch_one(call_id: str, api_key: str) -> dict:
    """Pull a single call by Vapi ID, normalize, validate it's Voicsito."""
    headers = {"Authorization": f"Bearer {api_key}"}
    r = httpx.get(f"{VAPI_BASE}/call/{call_id}", headers=headers, timeout=30.0)
    r.raise_for_status()
    return _normalize(r.json())


def _load_api_key() -> str:
    """Read VAPI_API_KEY from config/.env or env var."""
    if os.environ.get("VAPI_API_KEY"):
        return os.environ["VAPI_API_KEY"]
    env_path = Path(__file__).resolve().parent.parent / "config" / ".env"
    if env_path.exists():
        for line in env_path.read_text().splitlines():
            if line.startswith("VAPI_API_KEY="):
                return line.split("=", 1)[1].strip()
    sys.exit("VAPI_API_KEY not found in env or config/.env")


def main() -> int:
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    g = ap.add_mutually_exclusive_group(required=True)
    g.add_argument("--id", help="single Vapi call id")
    g.add_argument("--last", type=int, help="N most recent Voicsito calls")
    g.add_argument("--since", help="ISO timestamp; pull all Voicsito calls since this time")
    ap.add_argument("--filter", action="append", default=[],
                    help="key=value filter; repeatable. Keys: biz_type, ended_reason, stage_reached")
    args = ap.parse_args()

    api_key = _load_api_key()

    if args.id:
        result = [fetch_one(args.id, api_key)]
    else:
        # Implemented in Task 3
        sys.exit("--last / --since not yet implemented")

    print(json.dumps(result, indent=2, default=str))
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

ALLOWED_FILTER_KEYS = {"biz_type", "ended_reason", "stage_reached"}


def parse_filters(raw_filters: list) -> dict:
    """Parse a list of 'key=value' strings into a dict.

    Raises ValueError on unknown keys so typos surface immediately.
    """
    out = {}
    for raw in raw_filters:
        if "=" not in raw:
            raise ValueError(f"filter '{raw}' is not in key=value form")
        key, value = raw.split("=", 1)
        key = key.strip()
        if key not in ALLOWED_FILTER_KEYS:
            raise ValueError(
                f"unknown filter key '{key}'. allowed: {sorted(ALLOWED_FILTER_KEYS)}"
            )
        out[key] = value.strip()
    return out


def fetch_list(
    api_key: str,
    limit: int | None = None,
    since: str | None = None,
    filters: dict | None = None,
) -> list:
    """Pull recent Voicsito calls, optionally bounded and filtered.

    Vapi `/call` list endpoint params used: `limit`, `createdAtGt`.
    """
    headers = {"Authorization": f"Bearer {api_key}"}
    params = {}
    if limit:
        # over-fetch since we'll filter to Voicsito assistants client-side
        params["limit"] = max(limit * 5, 50)
    if since:
        params["createdAtGt"] = since

    r = httpx.get(f"{VAPI_BASE}/call", headers=headers, params=params, timeout=30.0)
    r.raise_for_status()

    raw_calls = r.json()
    out = []
    for raw in raw_calls:
        try:
            norm = _normalize(raw)
        except ValueError:
            continue  # not a Voicsito call — skip silently

        if filters:
            if filters.get("biz_type") and norm.get("biz_type") != filters["biz_type"]:
                continue
            if filters.get("ended_reason") and norm.get("ended_reason") != filters["ended_reason"]:
                continue
            if filters.get("stage_reached") and norm.get("stage_reached") != filters["stage_reached"]:
                continue

        out.append(norm)
        if limit and len(out) >= limit:
            break
    return out
```

import hashlib
from pathlib import Path

import pytest
import yaml


def _write_yaml(path: Path, data: dict):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(yaml.safe_dump(data, sort_keys=False))


def _sha256(text: str) -> str:
    return hashlib.sha256(text.encode("utf-8")).hexdigest()


def test_clean_state_returns_zero_drift(tmp_path):
    from scripts.check_checklist_sync import scan

    prompts_dir = tmp_path / "src" / "agent" / "prompts"
    prompts_dir.mkdir(parents=True)
    p = prompts_dir / "stage1_inbound_texts.py"
    p.write_text("PROMPT = 'hello'\n")

    changes_dir = tmp_path / "voicsito" / "call-review" / "changes"
    changes_dir.mkdir(parents=True)
    _write_yaml(changes_dir / "CH-2026-05-22-001.yaml", {
        "id": "CH-2026-05-22-001",
        "prompt_file": "src/agent/prompts/stage1_inbound_texts.py",

#!/usr/bin/env python3
"""Verify every prompt file's current sha256 matches the most-recent
`voicsito/call-review/changes/CH-*.yaml` entry that references it.

Exits 0 if clean; exits 1 with a drift report if any prompt file changed
without a matching log entry (or the latest matching entry doesn't have
an `escape_reason` justifying the unmatched sha).

CLI:
    python scripts/check_checklist_sync.py
    python scripts/check_checklist_sync.py --repo-root /path/to/Voicsito
"""
import argparse
import hashlib
import sys
from pathlib import Path

import yaml

PROMPT_GLOB = "src/agent/prompts/*.py"
CHANGES_GLOB = "voicsito/call-review/changes/CH-*.yaml"


def _sha256_of(path: Path) -> str:
    h = hashlib.sha256()
    h.update(path.read_bytes())
    return h.hexdigest()


def _load_change(path: Path) -> dict:
    try:
        return yaml.safe_load(path.read_text()) or {}
    except yaml.YAMLError as e:
        return {"_parse_error": str(e), "_path": str(path)}


def scan(repo_root: Path) -> list:
    """Return a list of drift records. Empty list = clean."""
    repo_root = Path(repo_root)
    prompts = sorted(repo_root.glob(PROMPT_GLOB))
    prompts = [p for p in prompts if p.name != "__init__.py"]

    changes = sorted(repo_root.glob(CHANGES_GLOB))
    changes_by_prompt: dict[str, list] = {}
    for c in changes:
        data = _load_change(c)
        target = data.get("prompt_file")
        if not target:
            continue
        changes_by_prompt.setdefault(target, []).append((c, data))

    drifts = []
    for prompt in prompts:
        rel = prompt.relative_to(repo_root).as_posix()
        current_sha = _sha256_of(prompt)
        history = changes_by_prompt.get(rel, [])

        if not history:
            drifts.append({
                "prompt_file": rel,
                "current_sha256": current_sha,
                "reason": "no change entries reference this prompt file",
            })
            continue

        # Latest entry by filename sort (CH IDs are date-prefixed).
        latest_path, latest = history[-1]
        if latest.get("prompt_file_sha256_after") == current_sha:
            continue  # clean

        if (latest.get("escape_reason") or "").strip():
            continue  # latest entry justifies the mismatch

        drifts.append({
            "prompt_file": rel,
            "current_sha256": current_sha,
            "latest_change": latest.get("id") or latest_path.name,
            "latest_sha_after": latest.get("prompt_file_sha256_after"),
            "reason": "sha mismatch and no escape_reason on latest change",
        })
    return drifts


def main() -> int:
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--repo-root", type=Path,
                    default=Path(__file__).resolve().parent.parent)
    args = ap.parse_args()

    drifts = scan(args.repo_root)
    if not drifts:
        print("✓ all prompts in sync with change log")
        return 0

    print("✗ drift detected:")
    for d in drifts:
        print(f"  - {d['prompt_file']}: {d['reason']}")
        if d.get("latest_change"):
            print(f"      latest change: {d['latest_change']}")
            print(f"      logged sha: {d.get('latest_sha_after')}")
            print(f"      current sha: {d['current_sha256']}")
    return 1


if __name__ == "__main__":
    sys.exit(main())
```

#!/usr/bin/env python3
"""Regenerate INDEX.md views from the YAMLs in each call-review subdir.

Usage:
    python voicsito/call-review/render.py                # render all four
    python voicsito/call-review/render.py --only checklist
"""
import argparse
from collections import defaultdict
from pathlib import Path

import yaml


def _load_all(directory: Path, prefix: str) -> list:
    out = []
    for p in sorted(directory.glob(f"{prefix}-*.yaml")):
        data = yaml.safe_load(p.read_text()) or {}
        data["_path"] = p
        out.append(data)
    return out


def render_checklist(directory: Path) -> None:
    items = _load_all(directory, "CL")
    by_cat: dict[str, list] = defaultdict(list)
    for item in items:
        by_cat[item.get("category", "uncategorized")].append(item)

    lines = ["# Checklist", "",
             "_Generated by `render.py` — do not hand-edit._",
             "",
             f"Total: {len(items)} items.",
             ""]
    for cat in sorted(by_cat.keys()):
        lines.append(f"## {cat}")
        lines.append("")
        lines.append("| ID | Severity | Status | Question |")
        lines.append("|---|---|---|---|")
        for item in sorted(by_cat[cat], key=lambda i: i.get("id", "")):
            q = (item.get("question") or "").split("\n")[0].strip()
            lines.append(
                f"| {item.get('id', '?')} "
                f"| {item.get('severity', '?')} "
                f"| {item.get('status', '?')} "
                f"| {q} |"
            )
        lines.append("")
    (directory / "INDEX.md").write_text("\n".join(lines))


def render_changes(directory: Path) -> None:
    items = _load_all(directory, "CH")
    items.sort(key=lambda i: i.get("id", ""), reverse=True)  # newest first

    lines = ["# Changes", "",
             "_Generated by `render.py` — do not hand-edit._",
             "",
             f"Total: {len(items)} entries.",
             "",
             "| ID | When | Stage | Section | Hypothesis |",
             "|---|---|---|---|---|"]
    for it in items:
        hyp = (it.get("hypothesis") or "").split("\n")[0].strip()[:80]
        lines.append(
            f"| {it.get('id', '?')} "
            f"| {it.get('created', '?')} "
            f"| {it.get('stage', '?')} "
            f"| {it.get('section', '?')} "
            f"| {hyp} |"
        )
    (directory / "INDEX.md").write_text("\n".join(lines))


def render_observations(directory: Path) -> None:
    items = _load_all(directory, "OBS")
    items.sort(key=lambda i: i.get("id", ""), reverse=True)

    lines = ["# Observations", "",
             "_Generated by `render.py` — do not hand-edit._",
             "",
             f"Total: {len(items)} entries.",
             "",
             "| ID | When | Stage | Verdict | Fault | Confidence |",
             "|---|---|---|---|---|---|"]
    for it in items:
        lines.append(
            f"| {it.get('id', '?')} "
            f"| {it.get('created', '?')} "
            f"| {it.get('stage', '?')} "
            f"| {it.get('verdict', '?')} "
            f"| {it.get('fault_attribution', '?')} "
            f"| {it.get('confidence', '?')} |"
        )
    (directory / "INDEX.md").write_text("\n".join(lines))


def render_sessions(directory: Path) -> None:
    """Sessions are markdown files with YAML frontmatter; pull frontmatter only."""
    sessions = sorted(directory.glob("20*.md"), reverse=True)
    lines = ["# Sessions", "",
             "_Generated by `render.py` — do not hand-edit._",
             "",
             f"Total: {len(sessions)} sessions.",
             "",
             "| File | Mode | Calls reviewed | Findings | Changes |",
             "|---|---|---|---|---|"]
    for s in sessions:
        text = s.read_text()
        meta = {}
        if text.startswith("---"):
            end = text.find("\n---", 3)
            if end != -1:
                meta = yaml.safe_load(text[3:end]) or {}
        lines.append(
            f"| [{s.name}]({s.name}) "
            f"| {meta.get('mode', '?')} "
            f"| {len(meta.get('calls_reviewed') or [])} "
            f"| {len(meta.get('findings') or [])} "
            f"| {len(meta.get('changes') or [])} |"
        )
    (directory / "INDEX.md").write_text("\n".join(lines))


RENDERERS = {
    "checklist": render_checklist,
    "changes": render_changes,
    "observations": render_observations,
    "sessions": render_sessions,
}


def main() -> int:
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--only", choices=sorted(RENDERERS.keys()),
                    help="render only one of the four directories")
    args = ap.parse_args()

    base = Path(__file__).resolve().parent
    targets = [args.only] if args.only else list(RENDERERS.keys())
    for name in targets:
        d = base / name
        if not d.exists():
            print(f"skip {name}: directory does not exist")
            continue
        RENDERERS[name](d)
        print(f"✓ rendered {name}/INDEX.md")
    return 0


if __name__ == "__main__":
    import sys
    sys.exit(main())
```

#!/usr/bin/env python3
"""Snapshot the current prompt files into docs/reference-points/<date>-<reason>/.

Matches the existing pattern (see docs/reference-points/2026-05-11-pre-stage-split/).
Use before a "ship it" deploy so we can roll back if the change goes badly.

CLI:
    python scripts/snapshot_reference_point.py \\
        --reason recap-fix \\
        --note "Added recap step before booking ask per CL-010"
"""
import argparse
import shutil
import sys
from datetime import date
from pathlib import Path

PROMPT_FILES = [
    "src/agent/prompts/shared_persona.py",
    "src/agent/prompts/stage1_inbound_texts.py",
    "src/agent/prompts/stage2_qualify.py",
    "src/agent/prompts/stage3_book.py",
]


def create_snapshot(repo_root: Path, date: str, short_reason: str, long_reason: str) -> Path:
    repo_root = Path(repo_root)
    snap = repo_root / "docs" / "reference-points" / f"{date}-{short_reason}"
    snap.mkdir(parents=True, exist_ok=False)

    for rel in PROMPT_FILES:
        src = repo_root / rel
        if src.exists():
            dst = snap / Path(rel).name
            shutil.copy2(src, dst)

    restore_md = f"""# Reference point: {date} — {short_reason}

**Reason:** {long_reason}

## What this snapshot contains

Prompt files copied at the moment this snapshot was created:

"""
    for rel in PROMPT_FILES:
        if (repo_root / rel).exists():
            restore_md += f"- `{Path(rel).name}` (from `{rel}`)\n"
    restore_md += f"""

## To restore

```bash
cp docs/reference-points/{date}-{short_reason}/shared_persona.py src/agent/prompts/
cp docs/reference-points/{date}-{short_reason}/stage1_inbound_texts.py src/agent/prompts/
cp docs/reference-points/{date}-{short_reason}/stage2_qualify.py src/agent/prompts/
cp docs/reference-points/{date}-{short_reason}/stage3_book.py src/agent/prompts/
.venv-extract/bin/python scripts/push_squad.py
```

Verify by pulling the live Vapi assistant config and confirming the system message contents match these files.
"""
    (snap / "RESTORE.md").write_text(restore_md)
    return snap


def main() -> int:
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--reason", required=True,
                    help="short-kebab tag for the snapshot directory (e.g., recap-fix)")
    ap.add_argument("--note", required=True,
                    help="one-sentence description for RESTORE.md")
    ap.add_argument("--date", default=str(date.today()),
                    help="YYYY-MM-DD (defaults to today)")
    ap.add_argument("--repo-root", type=Path,
                    default=Path(__file__).resolve().parent.parent)
    args = ap.parse_args()

    snap = create_snapshot(args.repo_root, args.date, args.reason, args.note)
    print(f"✓ snapshot created at {snap}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

# call-review — Voicsito review session orchestrator

## When you enter this skill

Acknowledge in one sentence: "Entering call-review mode for <single-call|batch|direct-edit>." Then load the standing context and wait for the user's next move.

## Standing context — load at session start

Always read these (in this order):
1. `voicsito/call-review/checklist/CL-*.yaml` — the rubric (every item).
2. `voicsito/call-review/sessions/INDEX.md` — find the last 2 sessions; read their YAML frontmatter only.
3. Grep `voicsito/call-review/changes/CH-*.yaml` for any entry with `created` in the last 7 days. Build a one-line banner per matching change: `<id> | <stage> | <section> | <one-line hypothesis>`. Surface this banner ONCE at the start of the session. Do NOT re-read changelog entries on every turn.
4. Read the prompt files you expect to touch: `src/agent/prompts/{shared_persona,stage1_inbound_texts,stage2_qualify,stage3_book}.py`.

## Modes (dispatched from trigger phrasing)

### Single-call mode
Trigger: user names a Vapi call id, or "review this call", or "review call <id>".

1. Run: `python scripts/fetch_call.py --id <vapi-id>` and ingest the resulting JSON.
2. Surface a 1-line summary in chat: `<call_id> — <duration>s — <ended_reason> — stage_reached=<stage>`.
3. Wait for the user's notes. They will narrate from their Google Doc — paste-or-verbal both fine.
4. For each finding, you do one of:
   - tag against an existing `CL-NNN` and create an `observations/OBS-*.yaml`
   - propose a new `CL-NNN` if the pattern is new
   - propose a prompt edit (subject to freshness contract + atomic rule below)
5. At session end: write `voicsito/call-review/sessions/<ISO-timestamp>-<short-id>.md`, run `voicsito/call-review/render.py`, run `scripts/check_checklist_sync.py`.

### Batch mode
Trigger: "review last N calls" or "review calls filtered by <key>=<value>".

1. Run: `python scripts/fetch_call.py --last <N> --filter <k>=<v>`.
2. Surface a per-call 1-line summary. Do NOT auto-deep-dive any one call.
3. Look for cross-call patterns; capture them as observations with `pattern_type: cross_call`.
4. Single-call deep-dives within a batch are fine — just pivot to single-call mode for that call.

### Direct-edit mode
Trigger: any phrase asking to change a prompt section without referencing a call ("let's tighten the reveal", "change X to Y").

1. No call pull. Skip to the freshness contract.
2. Edit proceeds normally; observation entry uses `pattern_type: direct_edit` and `call_id: null`.

## Freshness contract — runs before EVERY prompt edit

You may NOT call the Edit tool on any file under `src/agent/prompts/` until you have done these in this order:

1. **Read the target file fresh.** Even if you "just" read it three turns ago. Re-Read immediately before staging the diff. The Edit tool requires this anyway; the skill restates it as a hard step.
2. **If the edit touches voice ID / tool ID / assistant ID / squad ID:** pull live Vapi config first. Do NOT trust documentation. Use `curl` with `Authorization: Bearer $VAPI_API_KEY` against `https://api.vapi.ai/{assistant|tool|phone-number|squad}` to confirm current values. (This is what the user pulled this morning when the docs claimed F2BOg7Ry3zTPV6Zd62Cd was a stock voice — it was the deleted Jon PVC.)
3. **Confirm a checklist update is planned for the same turn.** Either you are editing an existing `checklist/CL-*.yaml`, or you are creating a new one, or you can articulate why this edit qualifies for `escape_reason` (typo/whitespace/comment-only).

## Atomic rule — same turn, no exceptions

Every prompt-file Edit (under `src/agent/prompts/`) must be paired in the **same turn** with one of:

- a new or modified `voicsito/call-review/checklist/CL-*.yaml`, AND a new `voicsito/call-review/changes/CH-*.yaml` entry
- OR a new `voicsito/call-review/changes/CH-*.yaml` entry whose `escape_reason` field is non-empty AND describes a non-behavioral edit

**End-of-edit verification:** call `python scripts/check_checklist_sync.py`. If it exits non-zero, your work is incomplete — fix the drift before continuing.

**Escape valve scope:** `escape_reason` is acceptable ONLY for typos in comments, whitespace, file-header reformatting, dead-code removal, docstring polish. Any change to wording the LLM reads, gate logic, tool routing, conditional branches, or step ordering is behavior-affecting and MUST pair with a checklist update.

## Composition with other skills

- **`prompt-update-discipline`** — **MANDATORY** before any prompt edit. Invoke it directly; it enforces word-preservation hygiene and the snapshot-to-reference-points step.
- **`prompt-review`** — **OPTIONAL**. Invoke only when the user explicitly enters sales-thinking mode (e.g., "let's attack the reveal" or "let's Hormozi this"). Skip when the user wants a structural / mechanical edit.
- **`live-system-testing`** — DO NOT invoke during review sessions. Reviews are read-only on third-party APIs.

## Read-only safety (HARD prohibition)

During a review session you have access to real prospect data via Vapi / GHL APIs. You MUST NOT:

- POST/PUT/DELETE to GHL (no contact updates, no calendar mutations, no tag changes)
- POST/PUT/DELETE to Twilio
- POST/PUT/DELETE to Cal.com / any calendar provider
- Call the booking webhook in any non-`_test_dry_run: true` mode

The ONLY write operations allowed in a review session:
- Local file writes (YAMLs, prompt files, snapshots)
- Vapi assistant PATCH **only** on explicit user "ship it" — and that PATCH touches assistant configs (model/voice/prompt), never prospect records

## Reviewer disagreement protocol

When Jon and Waseem score the same call differently:
- Default tie-breaker is Waseem's read.
- BUT: capture both reads in `observations/OBS-*.yaml` `notes` field with reviewer names, mark `confidence: low`, and rewrite the checklist item's `how_to_spot_a_fail` to remove the ambiguity that produced the disagreement.

## "Ship it" flow

User says "ship it" or "push it" or "deploy". You must:

1. Run `python scripts/check_checklist_sync.py` — block on non-zero exit.
2. Run `python scripts/snapshot_reference_point.py --reason <kebab-short> --note "<one-sentence>"` — capture the pre-deploy state.
3. (Optional, recommended for behavior-changing edits) Run `.venv-tests/bin/python scripts/sim-call.py` against the canonical personas; block on any judge failure.
4. Run `.venv-extract/bin/python scripts/push_squad.py`.
5. Update the relevant `changes/CH-*.yaml` with `deployed: true`, `deployed_at: <ISO>`, `reference_point: docs/reference-points/<dir>/`.
6. Run `python voicsito/call-review/render.py` to refresh INDEX views.
7. Confirm to the user with the call IDs that drove this change + the verify-by note.

## End-of-session

Always:
1. Write the session log to `voicsito/call-review/sessions/<ISO-timestamp>-<short-id>.md` with YAML frontmatter (mode, calls_reviewed, findings, changes, deferred, follow_ups) + free-form narrative.
2. Run `python voicsito/call-review/render.py` — refresh all four INDEX files.
3. Run `python scripts/check_checklist_sync.py` — confirm clean exit. If dirty, surface the drift and fix before closing the session.
4. Brief one-line summary to user: "Session done. N findings, M changes shipped, K deferred. Drift check: clean."
````
